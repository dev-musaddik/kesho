# ===== KESHO .htaccess hardening additions =====
#
# Add the lines below INSIDE the existing # BEGIN WordPress block,
# or at the top of .htaccess BEFORE the WP block.
# Do NOT replace the existing WP rewrite block.
#
# ============================================================

# Disable directory browsing
Options -Indexes

# Disable server signature
ServerSignature Off

# Protect wp-config.php
<Files wp-config.php>
    Order allow,deny
    Deny from all
</Files>

# Protect .htaccess itself
<Files .htaccess>
    Order allow,deny
    Deny from all
</Files>

# Protect debug.log
<Files debug.log>
    Order allow,deny
    Deny from all
</Files>

# Protect readme files
<FilesMatch "^(readme|license|wp-config-sample|install)\.(html|txt|php)$">
    Order allow,deny
    Deny from all
</FilesMatch>

# Block XML-RPC entirely
<Files xmlrpc.php>
    Order allow,deny
    Deny from all
</Files>

# Block PHP execution in uploads (defence in depth)
<Directory "wp-content/uploads/">
    <FilesMatch "\.(php|phtml|php3|php4|php5|phps|pht|inc)$">
        Order allow,deny
        Deny from all
    </FilesMatch>
</Directory>

# Security headers
<IfModule mod_headers.c>
    Header set X-Content-Type-Options "nosniff"
    Header set X-Frame-Options "SAMEORIGIN"
    Header set Referrer-Policy "strict-origin-when-cross-origin"
    Header set Permissions-Policy "geolocation=(), microphone=(), camera=()"
    # Strict-Transport-Security is also set at Cloudflare — only enable here if not using Cloudflare
    # Header set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
</IfModule>

# Block author scans (?author=N user enumeration)
RewriteEngine On
RewriteCond %{QUERY_STRING} ^author=([0-9]*) [NC]
RewriteRule .* https://keshopro.com/? [L,R=301]

# Block common malicious requests
RewriteCond %{REQUEST_URI} (\.|%2e)(\.|%2e)(\/|%2f)
RewriteRule .* - [F,L]
RewriteCond %{QUERY_STRING} (eval\(|base64_) [NC]
RewriteRule .* - [F,L]

# Force HTTPS (backup for Cloudflare 'Always HTTPS')
RewriteCond %{HTTPS} !=on
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
